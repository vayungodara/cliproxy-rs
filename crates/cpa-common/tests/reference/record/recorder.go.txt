// Package recorder appends one JSON line per recorded call to $CPA_RECORD_DIR/<pid>.jsonl.
// It exists only in the throwaway /tmp copy used to generate cliproxy-rs fixtures.
package recorder

import (
	"crypto/sha256"
	"encoding/base64"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"sync"
	"unicode/utf8"

	"github.com/router-for-me/CLIProxyAPI/v8/internal/registry"
)

var (
	mu      sync.Mutex
	starts  []int
	lookups []map[string]any
	seen    = map[[32]byte]bool{}
	file    *os.File
)

func Caps(info *registry.ModelInfo) any {
	if info == nil {
		return nil
	}
	out := map[string]any{
		"id": info.ID, "type": info.Type, "user_defined": info.UserDefined,
		"support_configuration_update": info.SupportConfigurationUpdate,
		"max_completion_tokens":        info.MaxCompletionTokens,
	}
	if t := info.Thinking; t != nil {
		levels := t.Levels
		if levels == nil {
			levels = []string{}
		}
		out["thinking"] = map[string]any{"min": t.Min, "max": t.Max, "zero_allowed": t.ZeroAllowed, "dynamic_allowed": t.DynamicAllowed, "levels": levels}
	}
	return out
}

func Begin() {
	mu.Lock()
	if len(starts) == 0 {
		lookups = nil
	}
	starts = append(starts, len(lookups))
	mu.Unlock()
}

func Lookup(model, provider string, info *registry.ModelInfo) {
	mu.Lock()
	lookups = append(lookups, map[string]any{"model": model, "provider": provider, "info": Caps(info)})
	mu.Unlock()
}

// Str records bytes as a JSON string, or as {"b64": ...} when they are not valid UTF-8
// (encoding/json would replace invalid bytes with U+FFFD).
func Str(b []byte) any {
	if !utf8.Valid(b) {
		return map[string]any{"b64": base64.StdEncoding.EncodeToString(b)}
	}
	return string(b)
}

func Err(err error) any {
	if err == nil {
		return nil
	}
	return err.Error()
}

// bytesSafe re-encodes every invalid UTF-8 string as {"b64": ...}.
func bytesSafe(v any) any {
	switch x := v.(type) {
	case string:
		return Str([]byte(x))
	case map[string]any:
		out := make(map[string]any, len(x))
		for k, item := range x {
			out[k] = bytesSafe(item)
		}
		return out
	case []any:
		out := make([]any, len(x))
		for i, item := range x {
			out[i] = bytesSafe(item)
		}
		return out
	}
	return v
}

func End(fn string, in, out any) {
	mu.Lock()
	defer mu.Unlock()
	start := starts[len(starts)-1]
	starts = starts[:len(starts)-1]
	in, out = bytesSafe(in), bytesSafe(out)
	rec := map[string]any{"fn": fn, "in": in, "out": out, "lookups": append([]map[string]any{}, lookups[start:]...)}
	line, errMarshal := json.Marshal(rec)
	if errMarshal != nil {
		panic(errMarshal)
	}
	key := sha256.Sum256(line)
	if seen[key] {
		return
	}
	seen[key] = true
	if file == nil {
		dir := os.Getenv("CPA_RECORD_DIR")
		if dir == "" {
			return
		}
		f, errOpen := os.OpenFile(filepath.Join(dir, fmt.Sprintf("%d.jsonl", os.Getpid())), os.O_CREATE|os.O_WRONLY|os.O_APPEND, 0o644)
		if errOpen != nil {
			panic(errOpen)
		}
		file = f
	}
	_, _ = file.Write(append(line, '\n'))
}
